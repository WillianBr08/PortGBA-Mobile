cmake_minimum_required(VERSION 3.22)
project(EliteReduxMobile C)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBMGBA_ONLY OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DISABLE_FRONTENDS ON CACHE BOOL "" FORCE)
set(M_CORE_GBA ON CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
foreach(opt BUILD_QT BUILD_SDL BUILD_GL BUILD_GLES2 BUILD_GLES3 USE_DEBUGGERS USE_GDB_STUB USE_EDITLINE USE_FFMPEG USE_MINIZIP USE_LIBZIP USE_SQLITE3 USE_ELF USE_LUA USE_LZMA USE_DISCORD_RPC ENABLE_SCRIPTING)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_ZLIB ON CACHE BOOL "" FORCE)
set(USE_PNG ON CACHE BOOL "" FORCE)
set(USE_EPOXY OFF CACHE STRING "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig TRUE)
set(MGBA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/mgba" CACHE PATH "mGBA source directory")
add_subdirectory("${MGBA_SOURCE_DIR}" vendor/mgba EXCLUDE_FROM_ALL)
set(ELITE_ENGINE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/android/app/src/main/cpp/engine.c" CACHE FILEPATH "Engine wrapper source")
add_library(elite-engine STATIC "${ELITE_ENGINE_SOURCE}")
target_sources(elite-engine PRIVATE android/app/src/main/cpp/compatibility.c)
target_link_libraries(elite-engine PUBLIC mgba)
get_target_property(MGBA_DEFINITIONS mgba COMPILE_DEFINITIONS)
target_compile_definitions(elite-engine PUBLIC ${MGBA_DEFINITIONS})
target_include_directories(elite-engine PUBLIC android/app/src/main/cpp)
if(ANDROID)
  add_library(eliteredux SHARED android/app/src/main/cpp/jni.c)
  target_link_libraries(eliteredux PRIVATE elite-engine android log)
  target_link_options(eliteredux PRIVATE "-Wl,-z,max-page-size=16384")
endif()
